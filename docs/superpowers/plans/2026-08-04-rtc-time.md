# RTC + Real Time Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add CMOS RTC reading, `SYS_RTC`/`SYS_UPTIME` syscalls, and a `clock`/`date` that show wall-clock time.

**Architecture:** A new `drivers/rtc.c` reads CMOS regs via ports 0x70/0x71 (BCD or binary per status-B bit 2, century from 0x32). Two new syscalls (34/35) expose `struct aos_time` and uptime-seconds. `clock.c` shows HH:MM:SS + DD.MM.YYYY, a new `date` command prints the wall clock; `scripts/rtctest.py` verifies serial output.

**Tech Stack:** Freestanding C11, x86 i386 inb/outb, QEMU headless regression.

**Spec:** `docs/superpowers/specs/2026-08-04-rtc-time-design.md`

## Global Constraints

- Build with `make`; verification is `make test` plus boot-time serial output.
- Syscall numbers 34 (`SYS_RTC`) and 35 (`SYS_UPTIME`) — next free after 33.
- `struct aos_time { int year, month, day, hour, minute, second; }` — duplicated in kernel and programs headers per the existing `aos_msg` pattern.
- RTC read address written with NMI-mask bit (`reg | 0x80`); UIP (status A bit 7) polled with a bounded guard loop.
- `get_tick()` stays monotonic for sleep/yield; RTC is wall-clock only.
- No commits until each task's build + relevant test passes.

```c
#include "ports.h"
#include "rtc.h"

static unsigned char rtc_read(unsigned char reg) {
    outb(0x70, (unsigned char)(reg | 0x80));   // NMI masked
    return inb(0x71);
}

static unsigned int bcd2bin(unsigned char v) {
    return (unsigned int)((v >> 4) * 10 + (v & 0xF));
}

int rtc_get(struct aos_time *t) {
    unsigned char stb = rtc_read(0x0B);
    int binary = (stb & 0x04) != 0;            // status B bit 2
    unsigned int guard = 100000;
    while (rtc_read(0x0A) & 0x80) {            // status A bit 7 = UIP
        if (--guard == 0) return -1;
    }
    unsigned char sec = rtc_read(0x00);
    unsigned char min = rtc_read(0x02);
    unsigned char hr  = rtc_read(0x04);
    unsigned char day = rtc_read(0x07);
    unsigned char mon = rtc_read(0x08);
    unsigned char yr  = rtc_read(0x09);
    unsigned char cen = rtc_read(0x32);
    if (!binary) {
        sec = (unsigned char)bcd2bin(sec);
        min = (unsigned char)bcd2bin(min);
        hr  = (unsigned char)bcd2bin(hr);
        day = (unsigned char)bcd2bin(day);
        mon = (unsigned char)bcd2bin(mon);
        yr  = (unsigned char)bcd2bin(yr);
    }
    int year = (cen >= 20 && cen <= 99) ? (int)cen * 100 + (int)yr : 2000 + (int)yr;
    t->year = year; t->month = (int)mon; t->day = (int)day;
    t->hour = (int)hr; t->minute = (int)min; t->second = (int)sec;
    return 0;
}
```

- [ ] **Step 1: Write `scripts/rtctest.py`**

```python
#!/usr/bin/env python3
import os, re, subprocess, time

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
ISO = os.path.join(ROOT, "aos.iso")
SER = "/tmp/aos-rtc.log"

def wait_for(path, seconds=10):
    end = time.time() + seconds
    while time.time() < end:
        if os.path.exists(path): return
        time.sleep(0.05)
    raise RuntimeError("timeout waiting for " + path)

def main():
    try: os.unlink(SER)
    except FileNotFoundError: pass
    qemu = subprocess.Popen([
        "qemu-system-i386", "-m", "256", "-cdrom", ISO, "-display", "none",
        "-serial", "file:" + SER,
    ], stdin=subprocess.PIPE, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    try:
        time.sleep(4)
        qemu.stdin.write(b"date\n")
        qemu.stdin.flush()
        end = time.time() + 10
        log = ""
        while time.time() < end:
            time.sleep(0.5)
            with open(SER, "r", errors="replace") as f: log = f.read()
            if "KERNEL PANIC" in log:
                raise AssertionError("rtc/date triggered a kernel panic")
            if re.search(r"\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}", log): break
        if not re.search(r"\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}", log):
            raise AssertionError("date did not print wall-clock time; log tail:\n" + log[-500:])
        print("PASS: RTC wall-clock date command")
        return 0
    finally:
        qemu.terminate()
        try: qemu.wait(timeout=5)
        except subprocess.TimeoutExpired: qemu.kill()

if __name__ == "__main__":
    sys.exit(main())
```
